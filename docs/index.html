🚀 快速開始

# 安裝 Claude Code
npm install -g @anthropic-ai/claude-code

# 設定 Bedrock
export AWS_DEFAULT_REGION="us-west-2"
export CLAUDE_CODE_USE_BEDROCK=1

# 開始使用
claude -p "Hello from Bedrock!"
